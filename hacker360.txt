install_missing() {
    local missing=()
    for t in "$@"; do
        command -v "$t" &>/dev/null || missing+=("$t")
    done
    if [ ${#missing[@]} -gt 0 ]; then
        echo -e "${YELLOW}[*] Installing: ${missing[*]}${RESET}"
        sudo apt update -qq
        sudo apt install -y "${missing[@]}"
    fi
}

pause() { echo ""; read -rp "Press Enter to continue..."; }

get_target() {
    read -rp "Enter target (IP/URL/domain): " TARGET
    if [ -z "$TARGET" ]; then
        echo -e "${RED}[!] No target entered.${RESET}"; sleep 1; return 1
    fi
    return 0
}
